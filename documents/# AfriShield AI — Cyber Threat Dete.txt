# AfriShield AI — Cyber Threat Detection Engine
# Classifies user-reported security incidents and returns a severity
# rating, confidence score, and next-step guidance. See README.md for
# background on the pipeline and each module's role.

from __future__ import annotations

import logging
import re

from entity_extractor import extract_entities
from conversation_memory import memory as conversation_memory

from .config import (
    CONFIDENCE_THRESHOLD,
    ENTITY_FALLBACK_HINT,
    HIGH_HINT,
    MAX_CONFIDENCE,
    MEDIUM_HINT,
    MIN_THREAT_SCORE,
)
from .enums import IncidentType, Intent, Severity
from .intent_detector import _detect_intent, _STRONG_INCIDENT_RE, detect_user_intent
from .knowledge_base import ThreatKnowledgeBase
from .models import ThreatAnalysisResult
from .nlp_utils import _phrase_matches, _tokenize
from .response_generator import TemplateResponseGenerator, ThreatResponseGenerator
from .threat_database import THREAT_CATEGORIES
from .type_defs import CategoryMap

# Library code shouldn't call basicConfig — callers control verbosity, e.g.:
#   logging.basicConfig(level=logging.DEBUG)
logger = logging.getLogger("afrishield_ai.threat_detector")

NO_SIGNAL_REMEDIATION = (
    "No specific threat indicators found. If you suspect "
    "something is wrong, describe the exact symptoms "
    "(unexpected messages, pop-ups, transactions, etc.) "
    "for a more precise assessment."
)

LOW_CONFIDENCE_REMEDIATION = (
    "This appears to be a general security question rather "
    "than an active incident report. Describe the exact "
    "problem, including any impacted accounts, devices, "
    "or suspicious transactions, for a more accurate "
    "assessment."
)


class ThreatDetector:
    """Core cyber-threat classification engine.

    analyze() runs as a short pipeline:
        normalize -> intent -> entities -> category -> confidence
        -> severity -> references -> response -> result
    """

    def __init__(
        self,
        categories: CategoryMap | None = None,
        knowledge_base: ThreatKnowledgeBase | None = None,
        generator: ThreatResponseGenerator | None = None,
    ):
        self.categories = categories or THREAT_CATEGORIES
        self.knowledge_base = knowledge_base or ThreatKnowledgeBase()
        self.generator = generator or TemplateResponseGenerator()

    # ------------------------------------------------------------
    # Pipeline steps
    # ------------------------------------------------------------

    @staticmethod
    def _normalize(text: str) -> str:
        text = text.lower().strip()
        return re.sub(r"\s+", " ", text)

    def _score_category(self, tokens: list[str], keywords: dict[str, int]) -> tuple[int, list[str]]:
        score = 0
        matches: list[str] = []
        for keyword, weight in keywords.items():
            if _phrase_matches(tokens, keyword):
                score += weight
                matches.append(keyword)
        return score, matches

    def _score_categories(self, tokens: list[str]) -> tuple[str | None, int, list[str]]:
        """Score every category by keyword match. Returns (None, best_score, [])
        when the best match doesn't clear MIN_THREAT_SCORE — best_score is kept
        so `_entity_fallback` can use it as a floor rather than starting at 0."""
        best_category = None
        best_score = 0
        best_matches: list[str] = []

        for category, data in self.categories.items():
            score, matches = self._score_category(tokens, data["keywords"])
            if score > best_score:
                best_category = category
                best_score = score
                best_matches = list(dict.fromkeys(matches))

        if best_category is not None and best_score >= MIN_THREAT_SCORE:
            return best_category, best_score, best_matches
        return None, best_score, []

    @staticmethod
    def _entity_fallback(entities: dict, floor_score: int = 0) -> tuple[str | None, int, list[str]]:
        """Infer a category from extracted entities (amounts, OTPs, devices)
        when keyword matching found nothing usable."""
        hint = entities.get("severity_hint", 0)
        if hint < ENTITY_FALLBACK_HINT:
            return None, 0, []

        if entities.get("amounts") or entities.get("banks"):
            return "Financial Fraud", max(floor_score, int(hint / 10)), ["financial signal detected"]
        if entities.get("otp_codes_or_mentions"):
            return "Phishing Attack", max(floor_score, int(hint / 10)), ["otp detected"]
        if entities.get("devices"):
            return "Account/Device Compromise", max(floor_score, int(hint / 10)), ["device signal detected"]
        return "Financial Fraud", max(floor_score, int(hint / 15)), ["entity-derived signal"]

    @staticmethod
    def _confidence_from_score(score: int) -> float:
        if score <= 0:
            return 0.0
        return round(min(MAX_CONFIDENCE, 35 + score * 10), 1)

    def _calculate_confidence(self, best_score: int, entities: dict) -> float:
        confidence = self._confidence_from_score(best_score)
        severity_hint = float(entities.get("severity_hint", 0))
        combined = round(min(MAX_CONFIDENCE, (confidence * 0.65) + (severity_hint * 0.35)), 1)
        return max(confidence, combined)

    @staticmethod
    def _calculate_severity(
        base_severity: Severity,
        incident_type: IncidentType,
        confidence: float,
        best_matches: list[str],
        entities: dict,
    ) -> Severity:
        severity = base_severity
        if incident_type != IncidentType.SECURITY_QUESTION:
            if confidence >= CONFIDENCE_THRESHOLD and len(best_matches) >= 2:
                severity = Severity.from_rank(base_severity.rank + 1)

        hint = entities.get("severity_hint", 0)
        if hint >= HIGH_HINT:
            severity = Severity.from_rank(min(3, severity.rank + 2))
        elif hint >= MEDIUM_HINT:
            severity = Severity.from_rank(min(3, severity.rank + 1))

        return severity

    def _store_memory(
        self, user_id: str | None, text: str, entities: dict,
        user_intent: Intent, intent_conf: float, severity: Severity, confidence: float,
    ) -> None:
        if not user_id:
            return
        try:
            conversation_memory.add_message(
                user_id,
                text,
                {
                    "entities": entities,
                    "user_intent": user_intent,
                    "intent_confidence": intent_conf,
                    "severity": severity,
                    "confidence": confidence,
                },
            )
        except Exception as exc:
            # Memory persistence must never break analysis, but we still
            # want a trace of what failed when debugging.
            logger.debug("memory persistence failed for user_id=%s: %s", user_id, exc)

    # ------------------------------------------------------------
    # Result construction — one helper, all call sites pass only
    # what differs from the defaults.
    # ------------------------------------------------------------

    @staticmethod
    def _build_result(
        text: str,
        category: str,
        severity: Severity,
        confidence: float = 0.0,
        incident_type: IncidentType = IncidentType.SECURITY_QUESTION,
        user_intent: Intent = Intent.UNKNOWN,
        intent_confidence: float = 0.0,
        matched_keywords: list[str] | None = None,
        remediation: str = "",
        references: list[str] | None = None,
        generated_response: str = "",
        entities: dict | None = None,
    ) -> ThreatAnalysisResult:
        return ThreatAnalysisResult(
            text=text,
            category=category,
            severity=severity,
            confidence=confidence,
            incident_type=incident_type,
            user_intent=user_intent,
            intent_confidence=intent_confidence,
            matched_keywords=matched_keywords or [],
            remediation=remediation,
            references=references or [],
            generated_response=generated_response,
            entities=entities or {},
        )

    # ------------------------------------------------------------
    # Orchestration
    # ------------------------------------------------------------

    def analyze(self, text: str, user_id: str | None = None) -> ThreatAnalysisResult:
        if not text or not text.strip():
            return self._build_result(text=text, category="No Threat Detected", severity=Severity.LOW)

        normalized = self._normalize(text)
        tokens = _tokenize(normalized)
        logger.debug("normalize: %r", normalized)

        incident_type = _detect_intent(text)
        user_intent, intent_conf = detect_user_intent(text)
        logger.debug("intent: %s, %s (%.0f%%)", incident_type, user_intent, intent_conf)

        entities = extract_entities(text)
        logger.debug("entities: %s", entities)

        category, score, matches = self._score_categories(tokens)
        if category is None:
            category, score, matches = self._entity_fallback(entities, score)
        logger.debug("category: %s score=%s matches=%s", category, score, matches)

        if category is None:
            return self._build_result(
                text=text, category="General Security Question", severity=Severity.LOW,
                remediation=NO_SIGNAL_REMEDIATION,
            )

        category_data = self.categories[category]
        base_severity = category_data["base_severity"]

        confidence = self._calculate_confidence(score, entities)
        logger.debug("confidence: %s", confidence)

        # A security-question intent is informational unless the text also
        # contains a direct first-person / experiential incident signal.
        if incident_type == IncidentType.SECURITY_QUESTION and not _STRONG_INCIDENT_RE.search(normalized):
            return self._build_result(
                text=text, category="General Security Question", severity=Severity.LOW,
                user_intent=user_intent, intent_confidence=intent_conf,
                remediation=LOW_CONFIDENCE_REMEDIATION,
            )

        severity = self._calculate_severity(base_severity, incident_type, confidence, matches, entities)
        logger.debug("severity: %s", severity)

        references = self.knowledge_base.retrieve(category)
        response = self.generator.generate(
            text=text, category=category, severity=severity,
            references=references, remediation=category_data["remediation"],
        )
        logger.debug("response: %s", response)

        result = self._build_result(
            text=text,
            category=category,
            severity=severity,
            confidence=confidence,
            incident_type=incident_type,
            user_intent=user_intent,
            intent_confidence=intent_conf,
            matched_keywords=matches,
            remediation=category_data["remediation"],
            references=references,
            generated_response=response,
            entities=entities,
        )

        self._store_memory(user_id, text, entities, user_intent, intent_conf, severity, confidence)

        return result

    def detect_threat(self, text: str) -> tuple[str, str]:
        """Backward-compatible helper: returns (category, severity_label)."""
        result = self.analyze(text)
        severity_label = result.severity.value.split(" ", 1)[1]  # strip emoji
        return (result.category, severity_label)


# ------------------------------------------------------------
# Module-level convenience functions
# ------------------------------------------------------------

_default_detector = ThreatDetector()


def detect_threat(text: str) -> tuple[str, str]:
    return _default_detector.detect_threat(text)


def analyze_threat(text: str, user_id: str | None = None) -> ThreatAnalysisResult:
    return _default_detector.analyze(text, user_id=user_id)


def get_threat_response(text: str):
    """Returns (category, severity_display, confidence, generated_response).
    Kept for backward compatibility with earlier callers."""
    result = analyze_threat(text)
    return (
        result.category,
        result.severity.value,
        result.confidence,
        result.generated_response or result.remediation,
    )
